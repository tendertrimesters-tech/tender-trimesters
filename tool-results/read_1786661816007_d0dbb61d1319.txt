    35→/** Calculate total seconds of a meditation from its beats */
    36→function totalSeconds(m: Meditation): number {
    37→  let t = 0;
    38→  for (const b of m.beats) {
    39→    if (b.type === "pause" && b.pauseSeconds) t += b.pauseSeconds;
    40→    if (b.type === "speak" && b.text) {
    41→      // ~130 words per minute = ~0.462 seconds per word
    42→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    43→    }
    44→    if (b.type === "breath" && b.text) {
    45→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    46→    }
    47→    if (b.type === "affirmation" && b.text) {
    48→      // Affirmations are read more slowly with pauses
    49→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    50→    }
    51→  }
    52→  return t;
    53→}
    54→
    55→/** Approximate seconds for a single beat */
    56→function beatDuration(b: MeditationBeat): number {
    57→  if (b.type === "pause" && b.pauseSeconds) return b.pauseSeconds;
    58→  if (b.type === "speak" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    59→  if (b.type === "breath" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    60→  if (b.type === "affirmation" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    61→  return 0;
    62→}
    63→
    64→/** Format seconds as m:ss */
    65→function fmtTime(s: number): string {
    66→  const m = Math.floor(s / 60);
    67→  const sec = Math.floor(s % 60);
    68→  return `${m}:${sec.toString().padStart(2, "0")}`;
    69→}

   391→function MeditationPlayer({
   392→  meditation,
   393→  onClose,
   394→  onJournal,
   395→  onCompleted,
   396→}: {
   397→  meditation: Meditation;
   398→  onClose: () => void;
   399→  onJournal: () => void;
   400→  onCompleted: (medId: string, durSec: number) => void;
   401→}) {
   402→  const total = totalSeconds(meditation);
   403→  const [elapsed, setElapsed] = useState(0);
   404→  const [playing, setPlaying] = useState(false);
   405→  const [finished, setFinished] = useState(false);
   406→  const [breathPhase, setBreathPhase] = useState<"idle" | "in" | "hold" | "out">("idle");
   407→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   408→
   409→  // Calculate current beat index from elapsed time
   410→  let cumTime = 0;
   411→  let currentBeatIndex = 0;
   412→  for (let i = 0; i < meditation.beats.length; i++) {
   413→    cumTime += beatDuration(meditation.beats[i]);
   414→    if (elapsed < cumTime) {
   415→      currentBeatIndex = i;
   416→      break;
   417→    }
   418→    if (i === meditation.beats.length - 1) currentBeatIndex = i;
   419→  }
   420→  const currentBeat = meditation.beats[currentBeatIndex];
   421→  const progress = total > 0 ? Math.min(elapsed / total, 1) : 0;
   422→
   423→  // Timer
   424→  useEffect(() => {
   425→    if (playing && !finished) {
   426→      intervalRef.current = setInterval(() => {
   427→        setElapsed((prev) => {
   428→          const next = prev + 1;
   429→          if (next >= total) {
   430→            setPlaying(false);
   431→            setFinished(true);
   432→            onCompleted(meditation.id, total);
   433→            toast.success("Beautiful practice, mama. 🌿");
   434→            return total;
   435→          }
   436→          return next;
   437→        });
   438→      }, 1000);
   439→    }
   440→    return () => {
   441→      if (intervalRef.current) clearInterval(intervalRef.current);
   442→    };
   443→  }, [playing, finished, total, meditation.id, onCompleted]);
   444→
   445→  // Breathing animation synced to breath beats
   446→  useEffect(() => {
   447→    if (currentBeat?.type === "breath" && playing) {
   448→      if (currentBeat.breath === "in") setBreathPhase("in");
   449→      else if (currentBeat.breath === "hold") setBreathPhase("hold");
   450→      else if (currentBeat.breath === "out") setBreathPhase("out");
   451→    } else if (currentBeat?.type !== "breath") {
   452→      setBreathPhase("idle");
   453→    }
   454→  }, [currentBeatIndex, currentBeat, playing]);
   455→
   456→  function togglePlay() {
   457→    if (finished) {
   458→      setElapsed(0);
   459→      setFinished(false);
   460→      setPlaying(true);
   461→    } else {
   462→      setPlaying(!playing);
   463→    }
   464→  }
   465→
   466→  // Breathing circle animation scale
   467→  const breathScale =
   468→    breathPhase === "in" ? "scale-100" :
   469→    breathPhase === "hold" ? "scale-100" :
   470→    breathPhase === "out" ? "scale-75" :
   471→    "scale-90";
   472→
   473→  const style = categoryStyle(meditation.category);
   474→
   475→  return (
   476→    <div className="fixed inset-0 z-50 bg-gradient-cream flex flex-col">
   477→      {/* Top bar */}
   478→      <div className="flex items-center justify-between px-4 py-3">
   479→        <button onClick={onClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   480→          <ChevronLeft className="w-5 h-5 text-moss-deep" />
   481→        </button>
   482→        <div className="text-xs text-muted-foreground font-medium">{meditation.title}</div>
   483→        <button onClick={onClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   484→          <X className="w-5 h-5 text-moss-deep" />
   485→        </button>
   486→      </div>
   487→
   488→      {/* Main content */}
   489→      <div className="flex-1 flex flex-col items-center justify-center px-6 -mt-12">
   490→        <AnimatePresence mode="wait">
   491→          <motion.div
   492→            key={currentBeatIndex}
   493→            initial={{ opacity: 0, y: 8 }}
   494→            animate={{ opacity: 1, y: 0 }}
   495→            exit={{ opacity: 0, y: -8 }}
   496→            transition={{ duration: 0.6 }}
   497→            className="text-center max-w-sm mx-auto"
   498→          >
   499→            {/* Breathing circle */}
   500→            {currentBeat?.type === "breath" ? (
   501→              <div className="flex flex-col items-center gap-4 mb-8">
   502→                <div className={cn(
   503→                  "w-40 h-40 rounded-full bg-gradient-to-br flex items-center justify-center transition-transform duration-[4000ms] ease-in-out",
   504→                  style.gradient,
   505→                  breathScale
   506→                )}>
   507→                  <div className="text-center">
   508→                    <Wind className="w-6 h-6 text-foreground/60 mx-auto mb-1" />
   509→                    <div className="text-sm font-medium text-foreground/70">
   510→                      {currentBeat.breath === "in" && "Breathe in"}
   511→                      {currentBeat.breath === "hold" && "Hold"}
   512→                      {currentBeat.breath === "out" && "Breathe out"}
   513→                    </div>
   514→                  </div>
   515→                </div>
   516→                {currentBeat.text && (
   517→                  <p className="text-sm text-muted-foreground italic leading-relaxed">
   518→                    {currentBeat.text}
   519→                  </p>
   520→                )}
   521→              </div>
   522→            ) : currentBeat?.type === "affirmation" ? (
   523→              <div className="flex flex-col items-center gap-4 mb-8">
   524→                <div className={cn(
   525→                  "w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center",
   526→                  "from-blush/80 to-rose-100/80"
   527→                )}>
   528→                  <Sparkles className="w-6 h-6 text-rose-gold" />
   529→                </div>
   530→                <p className="font-script text-xl text-rose-gold leading-relaxed">
   531→                  &ldquo;{currentBeat.text}&rdquo;
   532→                </p>
   533→              </div>
   534→            ) : currentBeat?.type === "pause" ? (
   535→              <div className="flex flex-col items-center gap-4 mb-8">
   536→                <div className={cn(
   537→                  "w-28 h-28 rounded-full flex items-center justify-center transition-transform duration-[2000ms]",
   538→                  "bg-cream-deep/40"
   539→                )}>
   540→                  <div className="w-3 h-3 rounded-full bg-moss/30 animate-pulse-soft" />
   541→                </div>
   542→                <p className="text-xs text-muted-foreground/60 tracking-wider">
   543→                  {currentBeat.pauseSeconds}s pause
   544→                </p>
   545→              </div>
   546→            ) : (
   547→              <div className="flex flex-col items-center gap-4 mb-8">
   548→                <div className={cn(
   549→                  "w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center",
   550→                  style.gradient,
   551→                  "opacity-60"
   552→                )}>
   553→                  <Flower2 className="w-8 h-8 text-foreground/40" />
   554→                </div>
   555→              </div>
   556→            )}
   557→
   558→            {/* Spoken text */}
   559→            {currentBeat?.type === "speak" && currentBeat.text && (
   560→              <motion.p
   561→                key={`text-${currentBeatIndex}`}
   562→                initial={{ opacity: 0 }}
   563→                animate={{ opacity: 1 }}
   564→                transition={{ duration: 0.8 }}
   565→                className="font-serif text-lg text-moss-deep leading-relaxed"
   566→              >
   567→                {currentBeat.text}
   568→              </motion.p>
   569→            )}
   570→          </motion.div>
   571→        </AnimatePresence>
   572→      </div>
   573→
   574→      {/* Bottom controls */}
   575→      <div className="px-6 pb-8 pt-4 space-y-4">
   576→        {/* Progress bar */}
   577→        <div className="space-y-1.5">
   578→          <div className="h-1.5 bg-cream-deep rounded-full overflow-hidden">
   579→            <motion.div
   580→              className={cn("h-full rounded-full bg-gradient-to-r", style.gradient)}
   581→              initial={{ width: 0 }}
   582→              animate={{ width: `${progress * 100}%` }}
   583→              transition={{ duration: 0.3 }}
   584→            />
   585→          </div>
   586→          <div className="flex justify-between text-[10px] text-muted-foreground">
   587→            <span>{fmtTime(elapsed)}</span>
   588→            <span>{fmtTime(total)}</span>
   589→          </div>
   590→        </div>
   591→
   592→        {/* Controls */}
   593→        <div className="flex items-center justify-center gap-6">
   594→          <Button
   595→            variant="ghost"
   596→            size="icon"
   597→            onClick={onClose}
   598→            className="text-muted-foreground"
   599→          >
   600→            <ChevronLeft className="w-5 h-5" />
   601→          </Button>
   602→
   603→          <button
   604→            onClick={togglePlay}
   605→            className={cn(
   606→              "w-16 h-16 rounded-full bg-gradient-to-br flex items-center justify-center shadow-lg transition-all active:scale-95",
   607→              style.gradient
   608→            )}
   609→          >
   610→            {playing ? (
   611→              <Pause className="w-7 h-7 text-moss-deep" />
   612→            ) : (
   613→              <Play className="w-7 h-7 text-moss-deep ml-1" />
   614→            )}
   615→          </button>
   616→
   617→          {finished && (
   618→            <Button
   619→              onClick={onJournal}
   620→              className="rounded-full bg-gradient-blush text-moss-deep hover:opacity-90"
   621→            >
   622→              <BookOpen className="w-4 h-4 mr-2" />
   623→              Journal
   624→            </Button>
   625→          )}
   626→        </div>
   627→      </div>
   628→    </div>
   629→  );
   630→}
