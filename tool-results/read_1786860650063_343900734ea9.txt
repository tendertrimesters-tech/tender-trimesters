    35→/** Calculate total seconds of a meditation from its beats */
    36→function totalSeconds(m: Meditation): number {
    37→  let t = 0;
    38→  for (const b of m.beats) {
    39→    if (b.type === "pause" && b.pauseSeconds) t += b.pauseSeconds;
    40→    if (b.type === "speak" && b.text) {
    41→      // ~130 words per minute = ~0.462 seconds per word
    42→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    43→    }
    44→    if (b.type === "breath" && b.text) {
    45→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    46→    }
    47→    if (b.type === "affirmation" && b.text) {
    48→      // Affirmations are read more slowly with pauses
    49→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    50→    }
    51→  }
    52→  return t;
    53→}
    54→
    55→/** Approximate seconds for a single beat */
    56→function beatDuration(b: MeditationBeat): number {
    57→  if (b.type === "pause" && b.pauseSeconds) return b.pauseSeconds;
    58→  if (b.type === "speak" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    59→  if (b.type === "breath" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    60→  if (b.type === "affirmation" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    61→  return 0;
    62→}

    64→/** Format seconds as m:ss */
    65→function fmtTime(s: number): string {
    66→  const m = Math.floor(s / 60);
    67→  const sec = Math.floor(s % 60);
    68→  return `${m}:${sec.toString().padStart(2, "0")}`;
    69→}

   393→function MeditationPlayer({
   394→  meditation,
   395→  onClose,
   396→  onJournal,
   397→  onCompleted,
   398→}: {
   399→  meditation: Meditation;
   400→  onClose: () => void;
   401→  onJournal: () => void;
   402→  onCompleted: (medId: string, durSec: number) => void;
   403→}) {
   404→  const total = totalSeconds(meditation);
   405→  const [elapsed, setElapsed] = useState(0);
   406→  const [playing, setPlaying] = useState(false);
   407→  const [finished, setFinished] = useState(false);
   408→  const [breathPhase, setBreathPhase] = useState<"idle" | "in" | "hold" | "out">("idle");
   409→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   410→
   411→  // Calculate current beat index from elapsed time
   412→  let cumTime = 0;
   413→  let currentBeatIndex = 0;
   414→  for (let i = 0; i < meditation.beats.length; i++) {
   415→    cumTime += beatDuration(meditation.beats[i]);
   416→    if (elapsed < cumTime) {
   417→      currentBeatIndex = i;
   418→      break;
   419→    }
   420→    if (i === meditation.beats.length - 1) currentBeatIndex = i;
   421→  }
   422→  const currentBeat = meditation.beats[currentBeatIndex];
   423→  const progress = total > 0 ? Math.min(elapsed / total, 1) : 0;
   424→
   425→  // Timer
   426→  useEffect(() => {
   427→    if (playing && !finished) {
   428→      intervalRef.current = setInterval(() => {
   429→        setElapsed((prev) => {
   430→          const next = prev + 1;
   431→          if (next >= total) {
   432→            setPlaying(false);
   433→            setFinished(true);
   434→            onCompleted(meditation.id, total);
   435→            toast.success("Beautiful practice, mama. 🌿");
   436→            return total;
   437→          }
   438→          return next;
   439→        });
   440→      }, 1000);
   441→    }
   442→    return () => {
   443→      if (intervalRef.current) clearInterval(intervalRef.current);
   444→    };
   445→  }, [playing, finished, total, meditation.id, onCompleted]);
   446→
   447→  // Breathing animation synced to breath beats
   448→  useEffect(() => {
   449→    if (currentBeat?.type === "breath" && playing) {
   450→      if (currentBeat.breath === "in") setBreathPhase("in");
   451→      else if (currentBeat.breath === "hold") setBreathPhase("hold");
   452→      else if (currentBeat.breath === "out") setBreathPhase("out");
   453→    } else if (currentBeat?.type !== "breath") {
   454→      setBreathPhase("idle");
   455→    }
   456→  }, [currentBeatIndex, currentBeat, playing]);
   457→
   458→  function togglePlay() {
   459→    if (finished) {
   460→      setElapsed(0);
   461→      setFinished(false);
   462→      setPlaying(true);
   463→    } else {
   464→      setPlaying(!playing);
   465→    }
   466→  }
   467→
   468→  function handleClose() {
   469→    // Stop timer and reset before closing
   470→    setPlaying(false);
   471→    if (intervalRef.current) {
   472→      clearInterval(intervalRef.current);
   473→      intervalRef.current = null;
   474→    }
   475→    onClose();
   476→  }
   477→
   478→  // Breathing circle animation scale
   479→  const breathScale =
   480→    breathPhase === "in" ? "scale-100" :
   481→    breathPhase === "hold" ? "scale-100" :
   482→    breathPhase === "out" ? "scale-75" :
   483→    "scale-90";
   484→
   485→  const style = categoryStyle(meditation.category);
   486→
   487→  return (
   488→    <div className="fixed inset-0 z-50 bg-gradient-cream flex flex-col">
   489→      {/* Top bar */}
   490→      <div className="flex items-center justify-between px-4 py-3">
   491→        <button onClick={handleClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   492→          <ChevronLeft className="w-5 h-5 text-moss-deep" />
   493→        </button>
   494→        <div className="text-xs text-muted-foreground font-medium">{meditation.title}</div>
   495→        <button onClick={handleClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   496→          <X className="w-5 h-5 text-moss-deep" />
   497→        </button>
   498→      </div>
   499→
   500→      {/* Main content */}
   501→      <div className="flex-1 flex flex-col items-center justify-center px-6 -mt-12">
   502→        <AnimatePresence mode="wait">
   503→          <motion.div
   504→            key={currentBeatIndex}
   505→            initial={{ opacity: 0, y: 8 }}
   506→            animate={{ opacity: 1, y: 0 }}
   507→            exit={{ opacity: 0, y: -8 }}
   508→            transition={{ duration: 0.6 }}
   509→            className="text-center max-w-sm mx-auto"
   510→          >
   511→            {/* Breathing circle */}
   512→            {currentBeat?.type === "breath" ? (
   513→              <div className="flex flex-col items-center gap-4 mb-8">
   514→                <div className={cn(
   515→                  "w-40 h-40 rounded-full bg-gradient-to-br flex items-center justify-center transition-transform duration-[4000ms] ease-in-out",
   516→                  style.gradient,
   517→                  breathScale
   518→                )}>
   519→                  <div className="text-center">
   520→                    <Wind className="w-6 h-6 text-foreground/60 mx-auto mb-1" />
   521→                    <div className="text-sm font-medium text-foreground/70">
   522→                      {currentBeat.breath === "in" && "Breathe in"}
   523→                      {currentBeat.breath === "hold" && "Hold"}
   524→                      {currentBeat.breath === "out" && "Breathe out"}
   525→                    </div>
   526→                  </div>
   527→                </div>
   528→                {currentBeat.text && (
   529→                  <p className="text-sm text-muted-foreground italic leading-relaxed">
   530→                    {currentBeat.text}
   531→                  </p>
   532→                )}
   533→              </div>
   534→            ) : currentBeat?.type === "affirmation" ? (
   535→              <div className="flex flex-col items-center gap-4 mb-8">
   536→                <div className={cn(
   537→                  "w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center",
   538→                  "from-blush/80 to-rose-100/80"
   539→                )}>
   540→                  <Sparkles className="w-6 h-6 text-rose-gold" />
   541→                </div>
   542→                <p className="font-script text-xl text-rose-gold leading-relaxed">
   543→                  &ldquo;{currentBeat.text}&rdquo;
   544→                </p>
   545→              </div>
   546→            ) : currentBeat?.type === "pause" ? (
   547→              <div className="flex flex-col items-center gap-4 mb-8">
   548→                <div className={cn(
   549→                  "w-28 h-28 rounded-full flex items-center justify-center transition-transform duration-[2000ms]",
   550→                  "bg-cream-deep/40"
   551→                )}>
   552→                  <div className="w-3 h-3 rounded-full bg-moss/30 animate-pulse-soft" />
   553→                </div>
   554→                <p className="text-xs text-muted-foreground/60 tracking-wider">
   555→                  {currentBeat.pauseSeconds}s pause
   556→                </p>
   557→              </div>
   558→            ) : (
   559→              <div className="flex flex-col items-center gap-4 mb-8">
   560→                <div className={cn(
   561→                  "w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center",
   562→                  style.gradient,
   563→                  "opacity-60"
   564→                )}>
   565→                  <Flower2 className="w-8 h-8 text-foreground/40" />
   566→                </div>
   567→              </div>
   568→            )}
   569→
   570→            {/* Spoken text */}
   571→            {currentBeat?.type === "speak" && currentBeat.text && (
   572→              <motion.p
   573→                key={`text-${currentBeatIndex}`}
   574→                initial={{ opacity: 0 }}
   575→                animate={{ opacity: 1 }}
   576→                transition={{ duration: 0.8 }}
   577→                className="font-serif text-lg text-moss-deep leading-relaxed"
   578→              >
   579→                {currentBeat.text}
   580→              </motion.p>
   581→            )}
   582→          </motion.div>
   583→        </AnimatePresence>
   584→      </div>
   585→
   586→      {/* Bottom controls */}
   587→      <div className="px-6 pb-8 pt-4 space-y-4">
   588→        {/* Progress bar */}
   589→        <div className="space-y-1.5">
   590→          <div className="h-1.5 bg-cream-deep rounded-full overflow-hidden">
   591→            <motion.div
   592→              className={cn("h-full rounded-full bg-gradient-to-r", style.gradient)}
   593→              initial={{ width: 0 }}
   594→              animate={{ width: `${progress * 100}%` }}
   595→              transition={{ duration: 0.3 }}
   596→            />
   597→          </div>
   598→          <div className="flex justify-between text-[10px] text-muted-foreground">
   599→            <span>{fmtTime(elapsed)}</span>
   600→            <span>{fmtTime(total)}</span>
   601→          </div>
   602→        </div>
   603→
   604→        {/* Controls */}
   605→        <div className="flex items-center justify-center gap-6">
   606→          <Button
   607→            variant="ghost"
   608→            size="icon"
   609→            onClick={handleClose}
   610→            className="text-muted-foreground"
   611→          >
   612→            <ChevronLeft className="w-5 h-5" />
   613→          </Button>
   614→
   615→          <button
   616→            onClick={togglePlay}
   617→            className={cn(
   618→              "w-16 h-16 rounded-full bg-gradient-to-br flex items-center justify-center shadow-lg transition-all active:scale-95",
   619→              style.gradient
   620→            )}
   621→          >
   622→            {playing ? (
   623→              <Pause className="w-7 h-7 text-moss-deep" />
   624→            ) : (
   625→              <Play className="w-7 h-7 text-moss-deep ml-1" />
   626→            )}
   627→          </button>
   628→
   629→          {finished && (
   630→            <Button
   631→              onClick={onJournal}
   632→              className="rounded-full bg-gradient-blush text-moss-deep hover:opacity-90"
   633→            >
   634→              <BookOpen className="w-4 h-4 mr-2" />
   635→              Journal
   636→            </Button>
   637→          )}
   638→        </div>
   639→      </div>
   640→    </div>
   641→  );
   642→}
