    50→function beatIndexAtTime(meditation: Meditation, elapsedSec: number): number {
    51→  let cumTime = 0;
    52→  for (let i = 0; i < meditation.beats.length; i++) {
    53→    cumTime += beatDuration(meditation.beats[i]);
    54→    if (elapsedSec < cumTime) return i;
    55→  }
    56→  return meditation.beats.length - 1;
    57→}

    59→/** Format seconds as m:ss */
    60→function fmtTime(s: number): string {
    61→  const m = Math.floor(s / 60);
    62→  const sec = Math.floor(s % 60);
    63→  return `${m}:${sec.toString().padStart(2, "0")}`;
    64→}

   508→/** Pick a warm female voice from available system voices */
   509→function pickVoice(): SpeechSynthesisVoice | null {
   510→  const voices = window.speechSynthesis?.getVoices() || [];
   511→  const preferred = [
   512→    "Samantha", "Karen", "Moira", "Tessa", "Fiona",
   513→    "Google UK English Female", "Microsoft Zira",
   514→    "Google US English", "Amazon Polly Joanna",
   515→  ];
   516→  for (const name of preferred) {
   517→    const v = voices.find((v) => v.name.includes(name) && v.lang.startsWith("en"));
   518→    if (v) return v;
   519→  }
   520→  return (
   521→    voices.find((v) => v.lang.startsWith("en") && v.name.toLowerCase().includes("female")) ||
   522→    voices.find((v) => v.lang.startsWith("en")) ||
   523→    null
   524→  );

   563→  const uploadInputRef = useRef<HTMLInputElement>(null);
   564→  const [uploading, setUploading] = useState(false);
   565→
   566→  // Keep playingRef in sync
   567→  playingRef.current = playing;
   568→  ttsBeatIndexRef.current = ttsBeatIndex;
   569→
   570→  // Determine current beat and elapsed based on mode
   571→  const currentBeat = useAudio
   572→    ? meditation.beats[beatIndexAtTime(meditation, audioElapsed)]
   573→    : meditation.beats[ttsBeatIndex];
   574→
   575→  const currentBeatIndex = useAudio
   576→    ? beatIndexAtTime(meditation, audioElapsed)
   577→    : ttsBeatIndex;
   578→
   579→  const elapsed = useAudio ? audioElapsed : ((): number => {
   580→    let t = 0;
   581→    for (let i = 0; i < ttsBeatIndex; i++) t += beatDuration(meditation.beats[i]);
   582→    return t;
   583→  })();
   584→
   585→  const total = useAudio && audioDuration > 0 ? audioDuration : estimatedTotal;
   586→  const progress = total > 0 ? Math.min(elapsed / total, 1) : 0;
   587→
   588→  // ── Load audio file from IndexedDB ───────────────────────────────
   589→  useEffect(() => {
   590→    if (!hasRecording) return;
   591→    let cancelled = false;
   592→    (async () => {
   593→      const blob = await getMeditationAudio(meditation.id);
   594→      if (cancelled || !blob) return;
   595→      const url = URL.createObjectURL(blob);
   596→      objectUrlRef.current = url;
   597→      setAudioUrl(url);
   598→      const audio = new Audio(url);
   599→      audio.preload = "metadata";
   600→      audioRef.current = audio;
   601→      audio.addEventListener("loadedmetadata", () => {
   602→        if (!cancelled) setAudioDuration(audio.duration);
   603→      });
   604→      audio.addEventListener("ended", () => {
   605→        if (!cancelled) {
   606→          setPlaying(false);
   607→          setFinished(true);
   608→          playingRef.current = false;
   609→          onCompleted(meditation.id, audio.duration);
   610→          toast.success("Beautiful practice, mama. 🌿");
   611→        }
   612→      });
   613→      audio.addEventListener("timeupdate", () => {
   614→        if (!cancelled) setAudioElapsed(audio.currentTime);
   615→      });
   616→    })();
   617→    return () => {
   618→      cancelled = true;
   619→      audioRef.current?.pause();
   620→      if (objectUrlRef.current) URL.revokeObjectURL(objectUrlRef.current);
   621→    };
   622→  }, [hasRecording, meditation.id, onCompleted]);
   623→
   624→  // Load TTS voices
   625→  useEffect(() => {
   626→    const synth = window.speechSynthesis;
   627→    if (!synth) return;
   628→    const load = () => { if (synth.getVoices().length > 0) setTtsReady(true); };
   629→    load();
   630→    synth.addEventListener("voiceschanged", load);
   631→    return () => synth.removeEventListener("voiceschanged", load);
   632→  }, []);
   633→
   634→  // Breathing animation synced to breath beats
   635→  useEffect(() => {
   636→    if (currentBeat?.type === "breath" && playing) {
   637→      if (currentBeat.breath === "in") setBreathPhase("in");
   638→      else if (currentBeat.breath === "hold") setBreathPhase("hold");
   639→      else if (currentBeat.breath === "out") setBreathPhase("out");
   640→    } else if (currentBeat?.type !== "breath") {
   641→      setBreathPhase("idle");
   642→    }
   643→  }, [currentBeatIndex, currentBeat, playing]);
   644→
   645→  // Cleanup on unmount
   646→  useEffect(() => {
   647→    return () => {
   648→      window.speechSynthesis?.cancel();
   649→      if (pauseTimerRef.current) clearTimeout(pauseTimerRef.current);
   650→      audioRef.current?.pause();
   651→    };
   652→  }, []);
   653→
   654→  // ── TTS beat engine ───────────────────────────────────────────────
   655→  const speakBeat = useCallback(
   656→    (index: number) => {
   657→      const synth = window.speechSynthesis;
   658→      if (!synth || !playingRef.current) return;
   659→
   660→      const beat = meditation.beats[index];
   661→      if (!beat) {
   662→        setPlaying(false);
   663→        setFinished(true);
   664→        playingRef.current = false;
   665→        onCompleted(meditation.id, estimatedTotal);
   666→        toast.success("Beautiful practice, mama. 🌿");
   667→        return;
   668→      }
   669→
   670→      setTtsBeatIndex(index);
   671→      ttsBeatIndexRef.current = index;
   672→
   673→      if (beat.type === "pause") {
   674→        const dur = beat.pauseSeconds || 5;
   675→        pauseTimerRef.current = setTimeout(() => {
   676→          if (playingRef.current) speakBeat(index + 1);
   677→        }, dur * 1000);
   678→        return;
   679→      }
   680→      if (beat.type === "journal") { speakBeat(index + 1); return; }
   681→
   682→      const text = beat.text || "";
   683→      if (!text) {
   684→        const dur = beatDuration(beat);
   685→        pauseTimerRef.current = setTimeout(() => {
   686→          if (playingRef.current) speakBeat(index + 1);
   687→        }, dur * 1000);
   688→        return;
   689→      }
   690→
   691→      synth.cancel();
   692→      const utterance = new SpeechSynthesisUtterance(text);
   693→      utteranceRef.current = utterance;
   694→      const voice = pickVoice();
   695→      if (voice) utterance.voice = voice;
   696→      utterance.rate = beat.type === "affirmation" ? 0.78 : 0.82;
   697→      utterance.pitch = 1.05;
   698→      utterance.volume = 0.95;
   699→
   700→      utterance.onend = () => {
   701→        utteranceRef.current = null;
   702→        pauseTimerRef.current = setTimeout(() => {
   703→          if (playingRef.current) speakBeat(index + 1);
   704→        }, beat.type === "affirmation" ? 2500 : 800);
   705→      };
   706→      utterance.onerror = (e) => {
   707→        if (e.error !== "canceled") console.warn("TTS error:", e.error);
   708→        utteranceRef.current = null;
   709→      };
   710→      synth.speak(utterance);
   711→    },
   712→    [meditation, onCompleted, estimatedTotal]
   713→  );
   714→
   715→  // ── Play / Pause / Skip ───────────────────────────────────────────
   716→  function togglePlay() {
   717→    if (useAudio && audioRef.current) {
   718→      // ── Audio mode ──
   719→      if (finished) {
   720→        audioRef.current.currentTime = 0;
   721→        setAudioElapsed(0);
   722→        setFinished(false);
   723→      }
   724→      if (playing) {
   725→        audioRef.current.pause();
   726→        setPlaying(false);
   727→      } else {
   728→        audioRef.current.play();
   729→        setPlaying(true);
   730→      }
   731→    } else {
   732→      // ── TTS mode ──
   733→      const synth = window.speechSynthesis;
   734→      if (!synth) return;
   735→      if (finished) {
   736→        setTtsBeatIndex(0);
   737→        setFinished(false);
   738→        setPlaying(true);
   739→        playingRef.current = true;
   740→        synth.cancel();
   741→        if (pauseTimerRef.current) clearTimeout(pauseTimerRef.current);
   742→        setTimeout(() => speakBeat(0), 300);
   743→        return;
   744→      }
   745→      if (playing) {
   746→        synth.cancel();
   747→        if (pauseTimerRef.current) clearTimeout(pauseTimerRef.current);
   748→        setPlaying(false);
   749→        playingRef.current = false;
   750→      } else {
   751→        setPlaying(true);
   752→        playingRef.current = true;
   753→        speakBeat(ttsBeatIndex);
   754→      }
   755→    }
   756→  }
   757→
   758→  function skipTo(direction: "prev" | "next") {
   759→    if (useAudio && audioRef.current) {
   760→      const currentIdx = beatIndexAtTime(meditation, audioElapsed);
   761→      let targetIdx = direction === "next" ? currentIdx + 1 : currentIdx - 1;
   762→      targetIdx = Math.max(0, Math.min(targetIdx, meditation.beats.length - 1));
   763→      // Calculate the start time of the target beat
   764→      let t = 0;
   765→      for (let i = 0; i < targetIdx; i++) t += beatDuration(meditation.beats[i]);
   766→      audioRef.current.currentTime = t;
   767→      setAudioElapsed(t);
   768→      setFinished(false);
   769→    } else {
   770→      window.speechSynthesis?.cancel();
   771→      if (pauseTimerRef.current) clearTimeout(pauseTimerRef.current);
   772→      let next = direction === "next" ? ttsBeatIndex + 1 : ttsBeatIndex - 1;
   773→      next = Math.max(0, Math.min(next, meditation.beats.length - 1));
   774→      setTtsBeatIndex(next);
   775→      setFinished(false);
   776→      if (playingRef.current) speakBeat(next);
   777→    }
   778→  }
   779→
   780→  function handleClose() {
   781→    window.speechSynthesis?.cancel();
   782→    if (pauseTimerRef.current) clearTimeout(pauseTimerRef.current);
   783→    audioRef.current?.pause();
   784→    playingRef.current = false;
   785→    onClose();
   786→  }
   787→
   788→  // ── Upload handler (inside player) ────────────────────────────────
   789→  async function handlePlayerUpload(e: React.ChangeEvent<HTMLInputElement>) {
   790→    const file = e.target.files?.[0];
   791→    if (!file) return;
   792→    setUploading(true);
   793→    try {
   794→      await saveMeditationAudio(meditation.id, file, file.name);
   795→      toast.success("Recording saved — it will play next time you open this meditation 🎙️");
   796→      onRecordingChange();
   797→    } catch {
   798→      toast.error("Failed to save recording");
   799→    } finally {
   800→      setUploading(false);
   801→      if (uploadInputRef.current) uploadInputRef.current.value = "";
   802→    }
   803→  }
   804→
   805→  const style = categoryStyle(meditation.category);
   806→
   807→  return (
   808→    <div className="fixed inset-0 z-50 bg-gradient-cream flex flex-col">
   809→      {/* Top bar */}
   810→      <div className="flex items-center justify-between px-4 py-3">
   811→        <button onClick={handleClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   812→          <ChevronLeft className="w-5 h-5 text-moss-deep" />
   813→        </button>
   814→        <div className="text-center">
   815→          <div className="text-xs text-muted-foreground font-medium">{meditation.title}</div>
   816→          <div className="text-[10px] text-muted-foreground/60 mt-0.5">
   817→            Beat {currentBeatIndex + 1} of {spokenBeats}
   818→          </div>
   819→        </div>
   820→        <button onClick={handleClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   821→          <X className="w-5 h-5 text-moss-deep" />
   822→        </button>
   823→      </div>
   824→
   825→      {/* Mode indicator + upload */}
   826→      <div className="flex justify-center -mt-1 mb-2 gap-2">
   827→        {useAudio && audioUrl ? (
   828→          <div className="flex items-center gap-1.5 px-3 py-1 rounded-full bg-moss/10 text-[10px] text-moss-deep font-medium">
   829→            <Mic className="w-3 h-3" /> Your recording
   830→          </div>
   831→        ) : ttsReady ? (
   832→          <div className="flex items-center gap-1.5 px-3 py-1 rounded-full bg-cream-deep/40 text-[10px] text-muted-foreground">
   833→            <Volume2 className="w-3 h-3" /> Default voice
   834→          </div>
   835→        ) : null}
   836→        <button
   837→          onClick={() => uploadInputRef.current?.click()}
   838→          disabled={uploading}
   839→          className="flex items-center gap-1 px-2.5 py-1 rounded-full bg-cream-deep/40 hover:bg-cream-deep/60 text-[10px] text-muted-foreground transition-colors disabled:opacity-50"
   840→          title="Upload your voice recording"
   841→        >
   842→          <Upload className="w-3 h-3" />
   843→          {uploading ? "Saving..." : (useAudio ? "Replace" : "Upload voice")}
   844→        </button>
   845→        <input ref={uploadInputRef} type="file" accept="audio/*" className="hidden" onChange={handlePlayerUpload} />
   846→      </div>
   847→
   848→      {/* Main content — same visual for both modes */}
   849→      <div className="flex-1 flex flex-col items-center justify-center px-6 -mt-4">
   850→        <AnimatePresence mode="wait">
   851→          <motion.div
   852→            key={currentBeatIndex}
   853→            initial={{ opacity: 0, y: 8 }}
   854→            animate={{ opacity: 1, y: 0 }}
   855→            exit={{ opacity: 0, y: -8 }}
   856→            transition={{ duration: 0.6 }}
   857→            className="text-center max-w-sm mx-auto"
   858→          >
   859→            {currentBeat?.type === "breath" ? (
   860→              <div className="flex flex-col items-center gap-4 mb-8">
   861→                <motion.div
   862→                  className={cn("w-40 h-40 rounded-full bg-gradient-to-br flex items-center justify-center", style.gradient)}
   863→                  animate={
   864→                    breathPhase === "in" ? { scale: [0.75, 1.15], opacity: [0.6, 1] }
   865→                    : breathPhase === "out" ? { scale: [1.15, 0.75], opacity: [1, 0.6] }
   866→                    : breathPhase === "hold" ? { scale: 1.15, opacity: 1 }
   867→                    : { scale: 0.9, opacity: 0.7 }
   868→                  }
   869→                  transition={{ duration: breathPhase === "in" || breathPhase === "out" ? 4 : 2, ease: "easeInOut" }}
   870→                >
   871→                  <div className="text-center">
   872→                    <Wind className="w-6 h-6 text-foreground/60 mx-auto mb-1" />
   873→                    <div className="text-sm font-medium text-foreground/70">
   874→                      {currentBeat.breath === "in" && "Breathe in"}
   875→                      {currentBeat.breath === "hold" && "Hold"}
   876→                      {currentBeat.breath === "out" && "Breathe out"}
   877→                    </div>
   878→                  </div>
   879→                </motion.div>
   880→                {currentBeat.text && (
   881→                  <p className="text-sm text-muted-foreground italic leading-relaxed">{currentBeat.text}</p>
   882→                )}
   883→              </div>
   884→            ) : currentBeat?.type === "affirmation" ? (
   885→              <div className="flex flex-col items-center gap-4 mb-8">
   886→                <motion.div
   887→                  className="w-32 h-32 rounded-full bg-gradient-to-br from-blush/80 to-rose-100/80 flex items-center justify-center"
   888→                  animate={{ scale: [1, 1.05, 1] }}
   889→                  transition={{ duration: 3, repeat: Infinity, ease: "easeInOut" }}
   890→                >
   891→                  <Sparkles className="w-6 h-6 text-rose-gold" />
   892→                </motion.div>
   893→                <p className="font-script text-xl text-rose-gold leading-relaxed">
   894→                    &ldquo;{currentBeat.text}&rdquo;
   895→                </p>
   896→              </div>
   897→            ) : currentBeat?.type === "pause" ? (
   898→              <div className="flex flex-col items-center gap-4 mb-8">
   899→                <motion.div
   900→                  className="w-28 h-28 rounded-full bg-cream-deep/40 flex items-center justify-center"
   901→                  animate={{ scale: [1, 1.1, 1], opacity: [0.4, 0.8, 0.4] }}
   902→                  transition={{ duration: (currentBeat.pauseSeconds || 5) * 0.8, repeat: Infinity, ease: "easeInOut" }}
   903→                >
   904→                  <div className="w-3 h-3 rounded-full bg-moss/30 animate-pulse-soft" />
   905→                </motion.div>
   906→                <p className="text-xs text-muted-foreground/60 tracking-wider">{currentBeat.pauseSeconds}s pause</p>
   907→              </div>
   908→            ) : (
   909→              <div className="flex flex-col items-center gap-4 mb-8">
   910→                <motion.div
   911→                  className={cn("w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center opacity-60", style.gradient)}
   912→                  animate={playing ? { scale: [1, 1.04, 1], opacity: [0.5, 0.7, 0.5] } : {}}
   913→                  transition={{ duration: 4, repeat: Infinity, ease: "easeInOut" }}
   914→                >
   915→                  <Flower2 className="w-8 h-8 text-foreground/40" />
   916→                </motion.div>
   917→              </div>
   918→            )}
   919→
   920→            {currentBeat?.type === "speak" && currentBeat.text && (
   921→              <motion.p
   922→                key={`text-${currentBeatIndex}`}
   923→                initial={{ opacity: 0 }}
   924→                animate={{ opacity: 1 }}
   925→                transition={{ duration: 0.8 }}
   926→                className="font-serif text-lg text-moss-deep leading-relaxed"
   927→              >
   928→                {currentBeat.text}
   929→              </motion.p>
   930→            )}
   931→          </motion.div>
   932→        </AnimatePresence>
   933→      </div>
   934→
   935→      {/* Bottom controls */}
   936→      <div className="px-6 pb-8 pt-4 space-y-4">
   937→        <div className="space-y-1.5">
   938→          <div className="h-1.5 bg-cream-deep rounded-full overflow-hidden">
   939→            <motion.div
   940→              className={cn("h-full rounded-full bg-gradient-to-r", style.gradient)}
   941→              initial={{ width: 0 }}
   942→              animate={{ width: `${progress * 100}%` }}
   943→              transition={{ duration: 0.3 }}
   944→            />
   945→          </div>
   946→          <div className="flex justify-between text-[10px] text-muted-foreground">
   947→            <span>{fmtTime(elapsed)}</span>
   948→            <span>{fmtTime(total)}</span>
   949→          </div>
   950→        </div>
   951→
   952→        <div className="flex items-center justify-center gap-4">
   953→          <button
   954→            onClick={() => skipTo("prev")}
   955→            className="p-2.5 rounded-full hover:bg-cream-deep/50 transition-colors text-muted-foreground"
   956→          >
   957→            <SkipBack className="w-5 h-5" />
   958→          </button>
   959→
   960→          <button
   961→            onClick={togglePlay}
   962→            className={cn(
   963→              "w-16 h-16 rounded-full bg-gradient-to-br flex items-center justify-center shadow-lg transition-all active:scale-95",
   964→              style.gradient
   965→            )}
   966→          >
   967→            {playing ? (
   968→              <Pause className="w-7 h-7 text-moss-deep" />
   969→            ) : (
   970→              <Play className="w-7 h-7 text-moss-deep ml-1" />
   971→            )}
   972→          </button>
   973→
   974→          <button
   975→            onClick={() => skipTo("next")}
   976→            className="p-2.5 rounded-full hover:bg-cream-deep/50 transition-colors text-muted-foreground"
   977→          >
   978→            <SkipForward className="w-5 h-5" />
   979→          </button>
   980→
   981→          {finished ? (
   982→            <Button
   983→              onClick={onJournal}
   984→              className="rounded-full bg-gradient-blush text-moss-deep hover:opacity-90 px-4"
   985→            >
   986→              <BookOpen className="w-4 h-4" />
   987→            </Button>
   988→          ) : (
   989→            <button
   990→              onClick={handleClose}
   991→              className="p-2.5 rounded-full hover:bg-cream-deep/50 transition-colors text-muted-foreground"
   992→            >
   993→              <X className="w-[18px] h-[18px]" />
   994→            </button>
   995→          )}
   996→        </div>
   997→      </div>
   998→    </div>
   999→  );
  1000→}
