   707→function AuthDialog({
   708→  open,
   709→  onOpenChange,
   710→  mode,
   711→  setMode,
   712→  onSuccess,
   713→}: {
   714→  open: boolean;
   715→  onOpenChange: (v: boolean) => void;
   716→  mode: "signin" | "signup";
   717→  setMode: (m: "signin" | "signup") => void;
   718→  onSuccess: () => void;
   719→}) {

   722→  const [password, setPassword] = useState("");
   723→  const [loading, setLoading] = useState(false);
   724→  const [error, setError] = useState("");
   725→
   726→  async function submit() {
   727→    setError("");
   728→    setLoading(true);
   729→    try {
   730→      const result = await signIn("credentials", {
   731→        email,
   732→        password,
   733→        name,
   734→        mode,
   735→        redirect: false,
   736→      });
   737→      if (!result || result.error) {
   738→        throw new Error(result?.error || "Authentication failed");
   739→      }

   741→      onOpenChange(false);
   742→      // Force reload so SessionProvider picks up the new session
   743→      setTimeout(() => window.location.reload(), 200);
   744→    } catch (e: unknown) {
   745→      const err = e as Error;
   746→      setError(err.message || "Something went wrong");
   747→    } finally {
   748→      setLoading(false);
   749→    }
   750→  }
   751→
   752→  return (
   753→    <Dialog open={open} onOpenChange={(v) => { onOpenChange(v); if (!v) { setError(""); setName(""); setEmail(""); setPassword(""); } }}>
   754→      <DialogContent className="bg-card rounded-3xl max-w-md">
   755→        <DialogHeader>
   756→          <DialogTitle className="font-serif text-2xl text-moss-deep">
   757→            {mode === "signup" ? "Create your account" : "Welcome back, mama"}
   758→          </DialogTitle>
   759→          <DialogDescription>
   760→            {mode === "signup"
   761→              ? "Your private pregnancy journal starts here. Free forever."
   762→              : "Sign in to continue your journey."}
   763→          </DialogDescription>
   764→        </DialogHeader>
   765→        <div className="space-y-4 mt-2">
   766→          {mode === "signup" && (
   767→            <div>
   768→              <Label htmlFor="au-name">Your name</Label>
   769→              <Input id="au-name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Mama" className="mt-1.5 rounded-xl" />
   770→            </div>
   771→          )}
   772→          <div>
   773→            <Label htmlFor="au-email">Email</Label>

   775→          </div>
   776→          <div>
   777→            <Label htmlFor="au-pw">Password</Label>
