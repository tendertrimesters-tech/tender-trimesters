    35→    35→/** Calculate total seconds of a meditation from its beats */
    36→    36→function totalSeconds(m: Meditation): number {
    37→    37→  let t = 0;
    38→    38→  for (const b of m.beats) {
    39→    39→    if (b.type === "pause" && b.pauseSeconds) t += b.pauseSeconds;
    40→    40→    if (b.type === "speak" && b.text) {
    41→    41→      // ~130 words per minute = ~0.462 seconds per word
    42→    42→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    43→    43→    }
    44→    44→    if (b.type === "breath" && b.text) {
    45→    45→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    46→    46→    }
    47→    47→    if (b.type === "affirmation" && b.text) {
    48→    48→      // Affirmations are read more slowly with pauses
    49→    49→      t += Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    50→    50→    }
    51→    51→  }
    52→    52→  return t;
    53→    53→}
    54→    54→
    55→    55→/** Approximate seconds for a single beat */
    56→    56→function beatDuration(b: MeditationBeat): number {
    57→    57→  if (b.type === "pause" && b.pauseSeconds) return b.pauseSeconds;
    58→    58→  if (b.type === "speak" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    59→    59→  if (b.type === "breath" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 130);
    60→    60→  if (b.type === "affirmation" && b.text) return Math.ceil((b.text.split(/\s+/).length * 60) / 100) + 3;
    61→    61→  return 0;
    62→    62→}

    64→    64→/** Format seconds as m:ss */
    65→    65→function fmtTime(s: number): string {
    66→    66→  const m = Math.floor(s / 60);
    67→    67→  const sec = Math.floor(s % 60);
    68→    68→  return `${m}:${sec.toString().padStart(2, "0")}`;
    69→    69→}

   391→   391→function MeditationPlayer({
   392→   392→  meditation,
   393→   393→  onClose,
   394→   394→  onJournal,
   395→   395→  onCompleted,
   396→   396→}: {
   397→   397→  meditation: Meditation;
   398→   398→  onClose: () => void;
   399→   399→  onJournal: () => void;
   400→   400→  onCompleted: (medId: string, durSec: number) => void;
   401→   401→}) {
   402→   402→  const total = totalSeconds(meditation);
   403→   403→  const [elapsed, setElapsed] = useState(0);
   404→   404→  const [playing, setPlaying] = useState(false);
   405→   405→  const [finished, setFinished] = useState(false);
   406→   406→  const [breathPhase, setBreathPhase] = useState<"idle" | "in" | "hold" | "out">("idle");
   407→   407→  const intervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   408→   408→
   409→   409→  // Calculate current beat index from elapsed time
   410→   410→  let cumTime = 0;
   411→   411→  let currentBeatIndex = 0;
   412→   412→  for (let i = 0; i < meditation.beats.length; i++) {
   413→   413→    cumTime += beatDuration(meditation.beats[i]);
   414→   414→    if (elapsed < cumTime) {
   415→   415→      currentBeatIndex = i;
   416→   416→      break;
   417→   417→    }
   418→   418→    if (i === meditation.beats.length - 1) currentBeatIndex = i;
   419→   419→  }
   420→   420→  const currentBeat = meditation.beats[currentBeatIndex];
   421→   421→  const progress = total > 0 ? Math.min(elapsed / total, 1) : 0;
   422→   422→
   423→   423→  // Timer
   424→   424→  useEffect(() => {
   425→   425→    if (playing && !finished) {
   426→   426→      intervalRef.current = setInterval(() => {
   427→   427→        setElapsed((prev) => {
   428→   428→          const next = prev + 1;
   429→   429→          if (next >= total) {
   430→   430→            setPlaying(false);
   431→   431→            setFinished(true);
   432→   432→            onCompleted(meditation.id, total);
   433→   433→            toast.success("Beautiful practice, mama. 🌿");
   434→   434→            return total;
   435→   435→          }
   436→   436→          return next;
   437→   437→        });
   438→   438→      }, 1000);
   439→   439→    }
   440→   440→    return () => {
   441→   441→      if (intervalRef.current) clearInterval(intervalRef.current);
   442→   442→    };
   443→   443→  }, [playing, finished, total, meditation.id, onCompleted]);
   444→   444→
   445→   445→  // Breathing animation synced to breath beats
   446→   446→  useEffect(() => {
   447→   447→    if (currentBeat?.type === "breath" && playing) {
   448→   448→      if (currentBeat.breath === "in") setBreathPhase("in");
   449→   449→      else if (currentBeat.breath === "hold") setBreathPhase("hold");
   450→   450→      else if (currentBeat.breath === "out") setBreathPhase("out");
   451→   451→    } else if (currentBeat?.type !== "breath") {
   452→   452→      setBreathPhase("idle");
   453→   453→    }
   454→   454→  }, [currentBeatIndex, currentBeat, playing]);
   455→   455→
   456→   456→  function togglePlay() {
   457→   457→    if (finished) {
   458→   458→      setElapsed(0);
   459→   459→      setFinished(false);
   460→   460→      setPlaying(true);
   461→   461→    } else {
   462→   462→      setPlaying(!playing);
   463→   463→    }
   464→   464→  }
   465→   465→
   466→   466→  // Breathing circle animation scale
   467→   467→  const breathScale =
   468→   468→    breathPhase === "in" ? "scale-100" :
   469→   469→    breathPhase === "hold" ? "scale-100" :
   470→   470→    breathPhase === "out" ? "scale-75" :
   471→   471→    "scale-90";
   472→   472→
   473→   473→  const style = categoryStyle(meditation.category);
   474→   474→
   475→   475→  return (
   476→   476→    <div className="fixed inset-0 z-50 bg-gradient-cream flex flex-col">
   477→   477→      {/* Top bar */}
   478→   478→      <div className="flex items-center justify-between px-4 py-3">
   479→   479→        <button onClick={onClose} className="p-2 rounded-full hover:bg-cream-deep/50 transition-colors">
   480→   480→          <ChevronLeft className="w-5 h-5 text-moss-deep" />
   481→   481→        </button>
   482→   482→        <div className="text-xs text-muted-foreground font-medium">{meditation.title}</div>

   484→   484→          <X className="w-5 h-5 text-moss-deep" />

   486→   486→      </div>
   487→   487→
   488→   488→      {/* Main content */}
   489→   489→      <div className="flex-1 flex flex-col items-center justify-center px-6 -mt-12">

   491→   491→          <motion.div
   492→   492→            key={currentBeatIndex}
   493→   493→            initial={{ opacity: 0, y: 8 }}
   494→   494→            animate={{ opacity: 1, y: 0 }}
   495→   495→            exit={{ opacity: 0, y: -8 }}
   496→   496→            transition={{ duration: 0.6 }}
   497→   497→            className="text-center max-w-sm mx-auto"
   498→   498→          >
   499→   499→            {/* Breathing circle */}
   500→   500→            {currentBeat?.type === "breath" ? (
   501→   501→              <div className="flex flex-col items-center gap-4 mb-8">
   502→   502→                <div className={cn(
   503→   503→                  "w-40 h-40 rounded-full bg-gradient-to-br flex items-center justify-center transition-transform duration-[4000ms] ease-in-out",
   504→   504→                  style.gradient,
   505→   505→                  breathScale

   507→   507→                  <div className="text-center">
   508→   508→                    <Wind className="w-6 h-6 text-foreground/60 mx-auto mb-1" />

   512→   512→                      {currentBeat.breath === "out" && "Breathe out"}
   513→   513→                    </div>
   514→   514→                  </div>
   515→   515→                </div>
   516→   516→                {currentBeat.text && (
   517→   517→                  <p className="text-sm text-muted-foreground italic leading-relaxed">
   518→   518→                    {currentBeat.text}
   519→   519→                  </p>
   520→   520→                )}
   521→   521→              </div>
   522→   522→            ) : currentBeat?.type === "affirmation" ? (
   523→   523→              <div className="flex flex-col items-center gap-4 mb-8">
   524→   524→                <div className={cn(
   525→   525→                  "w-32 h-32 rounded-full bg-gradient-to-br flex items-center justify-center",
   526→   526→                  "from-blush/80 to-rose-100/80"
   527→   527→                )}>
   528→   528→                  <Sparkles className="w-6 h-6 text-rose-gold" />
   529→   529→                </div>
   530→   530→                <p className="font-script text-xl text-rose-gold leading-relaxed">
   531→   531→                  &ldquo;{currentBeat.text}&rdquo;

   534→   534→            ) : currentBeat?.type === "pause" ? (
   535→   535→              <div className="flex flex-col items-center gap-4 mb-8">
   536→   536→                <div className={cn(
   537→   537→                  "w-28 h-28 rounded-full flex items-center justify-center transition-transform duration-[2000ms]",
   538→   538→                  "bg-cream-deep/40"
   539→   539→                )}>
   540→   540→                  <div className="w-3 h-3 rounded-full bg-moss/30 animate-pulse-soft" />
   541→   541→                </div>
   542→   542→                <p className="text-xs text-muted-foreground/60 tracking-wider">

   550→   550→                  style.gradient,
   551→   551→                  "opacity-60"

   554→   554→                </div>
   555→   555→              </div>
   556→   556→            )}
   557→   557→

   559→   559→            {currentBeat?.type === "speak" && currentBeat.text && (
   560→   560→              <motion.p
   561→   561→                key={`text-${currentBeatIndex}`}

   604→   604→            onClick={togglePlay}

   606→   606→              "w-16 h-16 rounded-full bg-gradient-to-br flex items-center justify-center shadow-lg transition-all active:scale-95",

   616→   616→
   617→   617→          {finished && (

   619→   619→              onClick={onJournal}
   620→   620→              className="rounded-full bg-gradient-blush text-moss-deep hover:opacity-90"
   621→   621→            >
   622→   622→              <BookOpen className="w-4 h-4 mr-2" />
   623→   623→              Journal
   624→   624→            </Button>
   625→   625→          )}
   626→   626→        </div>
